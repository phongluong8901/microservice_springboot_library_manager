--- lenh check folder, file trong du an, xem chuan Seo k
Get-ChildItem -Recurse | Where-Object { $_.FullName -notmatch '\\node_modules|\\.next|\\.git' } | ForEach-Object {
    $indent = "  " * ($_.FullName.Split('\').Count - (Get-Item .).FullName.Split('\').Count)
    if ($_.PSIsContainer) {
        "$indent📂 $($_.Name)"
    } else {
        "$indent📄 $($_.Name)"
    }
}
